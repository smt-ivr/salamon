// ================= משיכת הודעות והשמעה =================

async function loadMessages() {
    const token = state.userToken || localStorage.getItem('userToken');
    if (!token) {
        showMessage('alert-messages', 'אינך מחובר למערכת.', 'error');
        return;
    }

    const tbody = document.getElementById('messages-table-body');
    tbody.innerHTML = '<tr><td colspan="5" class="empty-state"><i class="fa-solid fa-circle-notch fa-spin"></i> מושך הודעות מהשרת...</td></tr>';
    setLoading('btn-refresh-messages', true, '<i class="fa-solid fa-rotate-right"></i>');

    try {
        const res = await fetch(`${API_BASE_URL}/messages/list`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ userToken: token }) // כבר לא שולחים נתיב לשרת!
        });
        
        const data = await res.json();
        setLoading('btn-refresh-messages', false, '<i class="fa-solid fa-rotate-right"></i> רענן נתונים');

        if (!res.ok) {
            showMessage('alert-messages', data.error || 'שגיאה בשליפת ההודעות', 'error');
            tbody.innerHTML = '<tr><td colspan="5" class="empty-state" style="color:red;">שגיאה בטעינה</td></tr>';
            return;
        }

        if (!data.messages || data.messages.length === 0) {
            tbody.innerHTML = '<tr><td colspan="5" class="empty-state">אין הודעות להצגה בשלוחה זו.</td></tr>';
            return;
        }

        tbody.innerHTML = '';
        data.messages.forEach(msg => {
            const tr = document.createElement('tr');
            const sizeKb = msg.size ? (msg.size / 1024).toFixed(1) + ' KB' : '-';
            
            // חילוץ המספר בלבד מתוך שם הקובץ (למשל: מ-"3653.wav" יישאר רק "3653")
            const fileId = msg.name.replace('.wav', '').replace('.mp3', '');
            
            tr.innerHTML = `
                <td dir="ltr" style="text-align:right; font-weight:600; color:var(--text-dark);">${msg.name}</td>
                <td dir="ltr" style="text-align:right;">${msg.mtime || '-'}</td>
                <td dir="ltr" style="text-align:right; color:var(--text-light);">${sizeKb}</td>
                <td>${msg.durationStr || '-'}</td>
                <td>
                    <button class="actions-btn" style="background:var(--secondary); color:white;" onclick="playAudio('${fileId}', '${msg.name}')">
                        <i class="fa-solid fa-play"></i> האזן
                    </button>
                </td>
            `;
            tbody.appendChild(tr);
        });

    } catch (err) {
        setLoading('btn-refresh-messages', false, '<i class="fa-solid fa-rotate-right"></i> רענן נתונים');
        showMessage('alert-messages', 'שגיאת רשת מול שרת ה-API', 'error');
        tbody.innerHTML = '<tr><td colspan="5" class="empty-state" style="color:red;">שגיאת תקשורת</td></tr>';
    }
}

async function playAudio(fileId, fileName) {
    const token = state.userToken || localStorage.getItem('userToken');
    if (!token) return alert("החיבור נותק, אנא התחבר מחדש");
    
    const playerWrapper = document.getElementById('audio-player-wrapper');
    const player = document.getElementById('main-audio-player');
    const title = document.getElementById('playing-title');
    
    playerWrapper.style.display = 'block';
    playerWrapper.classList.add('fade-in');
    title.innerHTML = `<i class="fa-solid fa-circle-notch fa-spin"></i> מתחבר לשרת... (בודק סטטוס)`;
    
    // הקישור מעודכן: שולחים רק fileId=3653 במקום path=ivr2...
    const streamUrl = `${API_BASE_URL}/messages/stream?fileId=${encodeURIComponent(fileId)}&userToken=${encodeURIComponent(token)}`;
    
    try {
        const checkRes = await fetch(streamUrl);
        
        if (!checkRes.ok) {
            const errorText = await checkRes.text();
            title.innerHTML = `<i class="fa-solid fa-triangle-exclamation" style="color:var(--danger);"></i> שגיאת שרת (${checkRes.status}): <span style="font-weight:normal;">${errorText}</span>`;
            return;
        }
        
        player.src = streamUrl;
        player.play().then(() => {
            title.innerHTML = `<i class="fa-solid fa-volume-high"></i> מנגן: <span dir="ltr">${fileName}</span>`;
        }).catch(e => {
            title.innerHTML = `<i class="fa-solid fa-triangle-exclamation" style="color:var(--danger);"></i> שגיאת הפעלה בדפדפן: ${e.message}`;
        });
        
    } catch (err) {
        title.innerHTML = `<i class="fa-solid fa-wifi" style="color:var(--danger);"></i> תקלת רשת מול השרת: ${err.message}`;
    }
}
